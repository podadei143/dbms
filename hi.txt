6. Procedure
CREATE OR REPLACE PROCEDURE calc_total_avg AS
    v_sum NUMBER;
    v_avg NUMBER;
BEGIN
    SELECT SUM(marks), AVG(marks)
    INTO v_sum, v_avg
    FROM student_marks;
    DBMS_OUTPUT.PUT_LINE('Total Marks: '||v_sum);
    DBMS_OUTPUT.PUT_LINE('Average Marks: '||ROUND(v_avg,2));
END;
/
SET SERVEROUTPUT ON;
BEGIN
    calc_total_avg;
END;
/
7. Function
CREATE OR REPLACE FUNCTION check_dept(p_deptno NUMBER)
RETURN BOOLEAN
IS
    v_count NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_count
    FROM DEPT
    WHERE DEPTNO=p_deptno;
    IF v_count>0 THEN
        RETURN TRUE;
    ELSE
        RETURN FALSE;
    END IF;
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error: '||SQLERRM);
        RETURN FALSE;
END;
/
DECLARE
    v_deptno NUMBER:=&deptno;
    v_result BOOLEAN;
BEGIN
    v_result:=check_dept(v_deptno);
    IF v_result THEN
        DBMS_OUTPUT.PUT_LINE('Department exists.');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Department does not exist.');
    END IF;
END;
/
8. Exception Handling
CREATE OR REPLACE FUNCTION display_employee(p_empno NUMBER)
RETURN BOOLEAN
IS
    v_ename EMP.ENAME%TYPE;
    v_job EMP.JOB%TYPE;
    v_sal EMP.SAL%TYPE;
    v_deptno EMP.DEPTNO%TYPE;
BEGIN
    SELECT ENAME, JOB, SAL, DEPTNO
    INTO v_ename, v_job, v_sal, v_deptno
    FROM EMP
    WHERE EMPNO=p_empno;
    DBMS_OUTPUT.PUT_LINE('Employee Number: '||p_empno);
    DBMS_OUTPUT.PUT_LINE('Employee Name: '||v_ename);
    DBMS_OUTPUT.PUT_LINE('Job: '||v_job);
    DBMS_OUTPUT.PUT_LINE('Salary: '||v_sal);
    DBMS_OUTPUT.PUT_LINE('Department No: '||v_deptno);
    RETURN TRUE;
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Employee not found.');
        RETURN FALSE;
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error: '||SQLERRM);
        RETURN FALSE;
END;
/
DECLARE
    v_empno NUMBER:=&empno;
    v_result BOOLEAN;
BEGIN
    v_result:=display_employee(v_empno);
END;

9. View & Trigger
CREATE OR REPLACE VIEW LoanDetails AS
SELECT C.Custid,
       C.Custname,
       NVL(H.HomeAmount,0)+NVL(V.VehicleAmount,0)
       AS TotalLoanAmount
FROM Customer C
LEFT JOIN (
    SELECT Custid, SUM(Amount) HomeAmount
    FROM HomeLoan
    GROUP BY Custid
) H ON C.Custid=H.Custid
LEFT JOIN (
    SELECT Custid, SUM(Amount) VehicleAmount
    FROM VehicleLoan
    GROUP BY Custid
) V ON C.Custid=V.Custid;
SELECT * FROM LoanDetails;
CREATE OR REPLACE TRIGGER DisplayHomeLoanDetails
AFTER INSERT ON HomeLoan
FOR EACH ROW
BEGIN
    DBMS_OUTPUT.PUT_LINE('New Home Loan Details');
    DBMS_OUTPUT.PUT_LINE('Loan ID: '||:NEW.HLoanid);
    DBMS_OUTPUT.PUT_LINE('Customer ID: '||:NEW.Custid);
    DBMS_OUTPUT.PUT_LINE('Loan Amount: '||:NEW.Amount);
END;
