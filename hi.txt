CREATE OR REPLACE FUNCTION check_dept(p_deptno NUMBER)
RETURN BOOLEAN
IS
    v_count NUMBER;
BEGIN
    SELECT COUNT(*) INTO v_count
    FROM DEPT
    WHERE DEPTNO=p_deptno;
    IF v_count>0 THEN
        RETURN TRUE;
    ELSE
        RETURN FALSE;
    END IF;
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error: '||SQLERRM);
        RETURN FALSE;
END;
/
DECLARE
    v_deptno NUMBER:=&deptno;
    v_result BOOLEAN;

    IF v_result THEN
        DBMS_OUTPUT.PUT_LINE('Department exists.');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Department does not exist.');
    END IF;
END;

CREATE OR REPLACE FUNCTION display_employee(p_empno NUMBER)
RETURN BOOLEAN
IS
    v_ename EMP.ENAME%TYPE;
    v_job EMP.JOB%TYPE;
    v_sal EMP.SAL%TYPE;
    v_deptno EMP.DEPTNO%TYPE;
BEGIN
    SELECT ENAME, JOB, SAL, DEPTNO
    INTO v_ename, v_job, v_sal, v_deptno
    FROM EMP
    WHERE EMPNO=p_empno;

    DBMS_OUTPUT.PUT_LINE('Employee Name: '||v_ename);

    DBMS_OUTPUT.PUT_LINE('Salary: '||v_sal);
    DBMS_OUTPUT.PUT_LINE('Department No: '||v_deptno);

        DBMS_OUTPUT.PUT_LINE('Employee not found.');
        RETURN FALSE;
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error: '||SQLERRM);
        RETURN FALSE;
END;

    FOREIGN KEY (Book_id) REFERENCES Book(Book_id)
);

CREATE TABLE Borrower (
    Borrower_ID NUMBER PRIMARY KEY,
    Borrower_Name VARCHAR2(30)

CREATE TABLE Book_Loans (
    Loan_ID NUMBER PRIMARY KEY,
    Borrower_ID NUMBER,
    Book_id NUMBER,
    Date_Out DATE,
    FOREIGN KEY (Borrower_ID) REFERENCES Borrower(Borrower_ID),
    FOREIGN KEY (Book_id) REFERENCES Book(Book_id)
);

-- a) Book details with publisher and author
SELECT B.Book_id,B.Title,P.Name,A.Author_Name
FROM Book B
JOIN Publisher P ON B.Publisher_Name=P.Name
JOIN Book_Authors A ON B.Book_id=A.Book_id;

-- b) Borrowers who borrowed more than 3 books
-- between January and June 2017
SELECT BR.Borrower_ID,BR.Borrower_Name,
COUNT(*) AS Total_Books
FROM Borrower BR
JOIN Book_Loans BL
ON BR.Borrower_ID=BL.Borrower_ID
WHERE BL.Date_Out >= DATE '2017-01-01'
AND BL.Date_Out < DATE '2017-07-01'
GROUP BY BR.Borrower_ID,BR.Borrower_Name
HAVING COUNT(*) > 3;

-- c) Delete a book and update related table
DELETE FROM Book_Authors WHERE Book_id=101;
DELETE FROM Book_Loans WHERE Book_id=101;
DELETE FROM Book WHERE Book_id=101;
```

FROM Salesman S
WHERE NOT EXISTS (
    SELECT 1 FROM Customer C
    WHERE C.Salesman_id=S.Salesman_id
    AND C.City=S.City
);

    v_id NUMBER := &eid;
    v_name Employee.Name%TYPE;
    v_basic NUMBER;
    v_da NUMBER;
    v_hra NUMBER;
    v_tax NUMBER;
    v_ded NUMBER;
    v_net NUMBER;
BEGIN
    SELECT Name,Basic_Pay
    INTO v_name,v_basic
    FROM Employee
    WHERE Id=v_id;

    v_da:=v_basic*0.10;
    v_hra:=v_basic*0.20;
    v_tax:=v_basic*0.05;
    v_ded:=v_basic*0.02;

    v_net:=v_basic+v_da+v_hra-v_tax-v_ded;

    DBMS_OUTPUT.PUT_LINE('EMPLOYEE PAYSLIP');
    DBMS_OUTPUT.PUT_LINE('Name: '||v_name);
    DBMS_OUTPUT.PUT_LINE('Basic Pay: '||v_basic);
    DBMS_OUTPUT.PUT_LINE('DA: '||v_da);
    DBMS_OUTPUT.PUT_LINE('HRA: '||v_hra);
    DBMS_OUTPUT.PUT_LINE('Tax: '||v_tax);
    DBMS_OUTPUT.PUT_LINE('Deduction: '||v_ded);
    DBMS_OUTPUT.PUT_LINE('Net Pay: '||v_net);
EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Employee not found');
END;
/
```
